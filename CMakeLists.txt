cmake_minimum_required(VERSION 3.16)
project(A_Last_Cup_of_Tea VERSION 3.0 LANGUAGES C CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(TEA_FETCH_FREEGLUT "Build the pinned FreeGLUT source" ON)
find_package(OpenGL REQUIRED)
if(TEA_FETCH_FREEGLUT)
  include(FetchContent)
  set(FREEGLUT_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
  set(FREEGLUT_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(FREEGLUT_BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
  set(FREEGLUT_REPLACE_GLUT ON CACHE BOOL "" FORCE)
  FetchContent_Declare(freeglut
    URL https://github.com/freeglut/freeglut/releases/download/v3.6.0/freeglut-3.6.0.tar.gz
    URL_HASH SHA256=9c3d4d6516fbfa0280edc93c77698fb7303e443c1aaaf37d269e3288a6c3ea52)
  FetchContent_MakeAvailable(freeglut)
  if(MINGW)
    # A static library does not need DLL version resources. Omitting that optional
    # resource also avoids windres's path-with-spaces bug in portable toolchains.
    get_target_property(glut_sources freeglut_static SOURCES)
    list(FILTER glut_sources EXCLUDE REGEX "\\.rc$")
    set_property(TARGET freeglut_static PROPERTY SOURCES "${glut_sources}")
  endif()
  set(TEA_GLUT freeglut_static)
else()
  find_package(GLUT REQUIRED)
  set(TEA_GLUT GLUT::GLUT)
endif()
add_executable(last_cup main.cpp LivingWorld.h LivingWorld.inl Homestead.h Homestead.inl)
set_source_files_properties(LivingWorld.inl Homestead.inl PROPERTIES HEADER_FILE_ONLY TRUE)
target_link_libraries(last_cup PRIVATE ${TEA_GLUT} OpenGL::GL OpenGL::GLU)
if(TEA_FETCH_FREEGLUT)
  target_compile_definitions(last_cup PRIVATE FREEGLUT_STATIC)
  target_include_directories(last_cup PRIVATE "${freeglut_SOURCE_DIR}/include")
endif()
if(MINGW)
  target_link_options(last_cup PRIVATE -static -static-libgcc -static-libstdc++)
endif()
if(MSVC)
  target_compile_options(last_cup PRIVATE /W4)
else()
  target_compile_options(last_cup PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME campaign_logic COMMAND last_cup --self-test)
